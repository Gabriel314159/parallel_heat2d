cmake_minimum_required(VERSION 3.18)
project(model_b_pressure_3d LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(MODELB_MPI "Build MPI/OpenMP executable" ON)
add_executable(model_b_serial model_b.cpp)
target_compile_options(model_b_serial PRIVATE -O3 -ffp-contract=off -Wall -Wextra -Wno-unknown-pragmas)
if(MODELB_MPI)
  find_package(MPI REQUIRED COMPONENTS CXX)
  find_package(OpenMP REQUIRED COMPONENTS CXX)
  add_executable(model_b_mpi model_b.cpp)
  target_compile_definitions(model_b_mpi PRIVATE USE_MPI)
  target_compile_options(model_b_mpi PRIVATE -O3 -ffp-contract=off -Wall -Wextra)
  target_link_libraries(model_b_mpi PRIVATE MPI::MPI_CXX OpenMP::OpenMP_CXX)
endif()
